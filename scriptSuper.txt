f_dow_verdict() =>
    int d = dayofweek
    d == dayofweek.monday    ? f_txt("🔵 Слабая гамма — жди ложных пробоев", "🔵 Слабая гамма", "🔵") :
     d == dayofweek.tuesday  ? f_txt("🔵 Умеренная — короткие цели", "🔵 Короткие цели", "🔵") :
     d == dayofweek.wednesday? f_txt("🟡 Тренд зарождается — набирает силу", "🟡 Зарождается", "🟡") :
     d == dayofweek.thursday ? f_txt("🟠 Сильная гамма — тренд работает", "🟠 Сильная гамма", "🟠") :

        step := 25.0

f_norm_pdf(float x) => (1.0 / math.sqrt(2.0 * 3.141592653589793)) * math.exp(-0.5 * x * x)
f_calc_gamma(float S, float K, float sigma, float T) =>
    if S <= 0 or K <= 0 or sigma <= 0 or T <= 0
        0.0
    else
        float d1 = (math.log(S / K) + (sigma * sigma * 0.5) * T) / (sigma * math.sqrt(T))
        f_norm_pdf(d1) / (S * sigma * math.sqrt(T))
f_calc_charm(float S, float K, float sigma, float T) =>
    if S <= 0 or K <= 0 or sigma <= 0 or T <= 0
        0.0
    else
        float d1 = (math.log(S / K) + (sigma * sigma * 0.5) * T) / (sigma * math.sqrt(T))
        float d2 = d1 - sigma * math.sqrt(T)
        -f_norm_pdf(d1) * d2 / (2.0 * T)
f_round_step(float pr) =>
    float s = pr >= 50000 ? 1000.0 : pr >= 10000 ? 500.0 : pr >= 5000 ? 100.0 : pr >= 2000 ? 50.0 : pr >= 1000 ? 25.0 : pr >= 100 ? 5.0 : pr >= 10 ? 1.0 : pr >= 1 ? 0.1 : 0.01
    s
f_avoid_round(float price, bool is_long, float atr_v) =>
    float step = f_round_step(price)
    float nr = math.round(price / step) * step
    float dist = math.abs(price - nr)
    float zone = atr_v * 0.35
    float result = price
    if dist < zone and step > 0
        float sh = math.max(atr_v * 0.12, syminfo.mintick * 5)
        result := is_long ? math.min(nr - sh, price) : math.max(nr + sh, price)
    result
f_avoid_round_sl(float price, bool is_long, float atr_v) =>
    float step = f_round_step(price)
    float nr = math.round(price / step) * step
    float dist = math.abs(price - nr)
    float zone = atr_v * 0.35
    float result = price
    if dist < zone and step > 0
        float sh = math.max(atr_v * 0.12, syminfo.mintick * 5)
        result := is_long ? nr - sh : nr + sh
    result

f_eval_tf_trend() =>
    float _c = close
    float _hh20 = ta.highest(high, 20)
    float _ll20 = ta.lowest(low, 20)
    float _e50 = ta.ema(close, 50)
    float _e200 = ta.ema(close, 200)
    float _rsi14 = ta.rsi(close, 14)
    float _range = _hh20 - _ll20
    float _eq = _range > 0 ? (_ll20 + _range * 0.50) : _c
    bool _bull = true
    if _c < _e200 and _c < _e50 and _rsi14 < 46.0
        _bull := false
    else if _c >= _e200 and (_c >= _e50 or _rsi14 >= 48.0 or _c >= _eq)
        _bull := true
    else if _c >= _e50 and _rsi14 >= 48.0
        _bull := true
    else if _c < _e50 and _rsi14 < 48.0 and _c < _eq
        _bull := false
    else
        _bull := _c >= _e50 or _rsi14 >= 50.0
    _bull
